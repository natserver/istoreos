<% local function vue_lang() local i18n = require("luci.i18n") local lang = i18n.translate("quickstart_vue_lang") if lang == "quickstart_vue_lang" or lang == "" then -- 翻译目录没加载时按 LuCI 当前语言回落。 -- quickstart 的 SPA 是「以中文为源语言」的： -- i18n/zh-cn.json 是空壳，翻译表为空时回退到 key，而 key 就是 -- 中文原文；只有 i18n/en.json 存着「中文 key -> 英文」对照表。 -- 依次按 UCI 配置 -> i18n 上下文 -> 浏览器头 判断，zh* 映射成 zh-cn。 local cur = "" local ok_uci, uci = pcall(require, "luci.model.uci") if ok_uci and uci and uci.cursor then local ok_cur, c = pcall(uci.cursor) if ok_cur and c then cur = c:get("luci", "main", "lang") or "" end end if cur == "" or cur == "auto" then cur = (i18n.context and i18n.context.lang) or "" end if cur == "" or cur == "auto" then local ok_http, http = pcall(require, "luci.http") if ok_http and http and http.getenv then cur = http.getenv("HTTP_ACCEPT_LANGUAGE") or "" end end if tostring(cur):lower():match("^zh") then lang = "zh-cn" else lang = "en" end end return lang end -%> <% luci.template.render("quickstart/main", {prefix=luci.dispatcher.build_url("admin", "quickstart"),lang=vue_lang()}) %>
